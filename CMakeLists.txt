cmake_minimum_required(VERSION 3.20)
project(MoonRiverStudio VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(mrs_core
  core/src/model.cpp core/src/timeline.cpp core/src/transport.cpp
  core/src/project_store.cpp core/src/serialization.cpp core/src/fixtures.cpp)
target_include_directories(mrs_core PUBLIC core/include)
target_compile_features(mrs_core PUBLIC cxx_std_20)
set_target_properties(mrs_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(mrs_core PUBLIC /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mrs_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(mrs_core_check tools/core_check.cpp)
target_link_libraries(mrs_core_check PRIVATE mrs_core)
if(BUILD_TESTING)
  add_executable(mrs_core_tests tests/core_tests.cpp)
  target_link_libraries(mrs_core_tests PRIVATE mrs_core)
  foreach(suite model timeline transport commands serialization integration)
    add_test(NAME core_${suite} COMMAND mrs_core_tests ${suite})
  endforeach()
  add_test(NAME core_acceptance COMMAND mrs_core_check)
endif()
