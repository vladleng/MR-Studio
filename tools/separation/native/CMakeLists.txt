cmake_minimum_required(VERSION 3.20)
project(MRNativeSeparationProbe LANGUAGES CXX)
enable_testing()
add_executable(mrs_native_sw_frontend_test sw_frontend_test.cpp)
target_compile_features(mrs_native_sw_frontend_test PRIVATE cxx_std_20)
add_test(NAME native_sw_frontend COMMAND mrs_native_sw_frontend_test)
if(MSVC)
    target_compile_options(mrs_native_sw_frontend_test PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(mrs_native_sw_frontend_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set(MRS_AUDIOCPP_SDK_ROOT "" CACHE PATH "Verified audio.cpp v0.9.3 C API SDK")
find_path(AUDIOCPP_INCLUDE audiocpp.h PATHS "${MRS_AUDIOCPP_SDK_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(AUDIOCPP_LIBRARY NAMES audiocpp PATHS "${MRS_AUDIOCPP_SDK_ROOT}/libs" "${MRS_AUDIOCPP_SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(mrs_native_separation_probe probe.cpp ../../../core/src/sha256.cpp)
target_compile_features(mrs_native_separation_probe PRIVATE cxx_std_20)
target_include_directories(mrs_native_separation_probe SYSTEM PRIVATE "${AUDIOCPP_INCLUDE}")
target_include_directories(mrs_native_separation_probe PRIVATE ../../../core/include)
target_link_libraries(mrs_native_separation_probe PRIVATE "${AUDIOCPP_LIBRARY}")
option(MRS_AUDIOCPP_PROGRESS_TESTS "Test the custom Demucs chunk-progress C API extension" OFF)
set(MRS_AUDIOCPP_PROGRESS_MODEL "" CACHE FILEPATH "Verified HTDemucs6 GGUF for extension regression")
if(MRS_AUDIOCPP_PROGRESS_TESTS)
    if(NOT EXISTS "${MRS_AUDIOCPP_PROGRESS_MODEL}")
        message(FATAL_ERROR "Progress regression requires the explicitly downloaded verified HTDemucs6 GGUF")
    endif()
    add_executable(mrs_native_capi_progress_test capi_progress_test.cpp)
    target_compile_features(mrs_native_capi_progress_test PRIVATE cxx_std_20)
    target_include_directories(mrs_native_capi_progress_test SYSTEM PRIVATE "${AUDIOCPP_INCLUDE}")
    target_link_libraries(mrs_native_capi_progress_test PRIVATE "${AUDIOCPP_LIBRARY}")
    if(MSVC)
        target_compile_options(mrs_native_capi_progress_test PRIVATE /W4 /WX /permissive- /utf-8)
        target_compile_definitions(mrs_native_capi_progress_test PRIVATE AUDIOCPP_USE_DLL)
    else()
        target_compile_options(mrs_native_capi_progress_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME native_capi_progress COMMAND mrs_native_capi_progress_test "${MRS_AUDIOCPP_PROGRESS_MODEL}")
    set_tests_properties(native_capi_progress PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
endif()
if(MSVC)
    target_compile_options(mrs_native_separation_probe PRIVATE /W4 /WX /permissive- /utf-8)
    target_link_libraries(mrs_native_separation_probe PRIVATE psapi)
    target_compile_definitions(mrs_native_separation_probe PRIVATE AUDIOCPP_USE_DLL NOMINMAX WIN32_LEAN_AND_MEAN)
    file(GLOB native_dlls "${MRS_AUDIOCPP_SDK_ROOT}/libs/*.dll" "${MRS_AUDIOCPP_SDK_ROOT}/lib/*.dll")
    foreach(dll IN LISTS native_dlls)
        add_custom_command(TARGET mrs_native_separation_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:mrs_native_separation_probe>")
    endforeach()
else()
    target_compile_options(mrs_native_separation_probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
