cmake_minimum_required(VERSION 3.20)
project(MRTimestretchDrumsSDK VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_library(mr_timestretch_drums STATIC src/stretch.cpp)
add_library(MRTimestretchDrums::SDK ALIAS mr_timestretch_drums)
target_compile_features(mr_timestretch_drums PUBLIC cxx_std_20)
target_include_directories(mr_timestretch_drums PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(mr_timestretch_drums PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mr_timestretch_drums PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(mr_timestretch_drums_tests tests/tests.cpp)
  target_link_libraries(mr_timestretch_drums_tests PRIVATE MRTimestretchDrums::SDK)
  add_test(NAME drums_sdk_contract COMMAND mr_timestretch_drums_tests)
  set_tests_properties(drums_sdk_contract PROPERTIES TIMEOUT 120)
endif()
include(GNUInstallDirs)
install(TARGETS mr_timestretch_drums EXPORT MRTimestretchDrumsTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MRTimestretchDrumsTargets FILE MRTimestretchDrumsTargets.cmake NAMESPACE MRTimestretchDrums::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MRTimestretchDrums)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/mr-timestretch-drums)

set_target_properties(mr_timestretch_drums PROPERTIES EXPORT_NAME SDK)
add_executable(mr_timestretch_drums_example examples/render_clip.cpp)
target_link_libraries(mr_timestretch_drums_example PRIVATE MRTimestretchDrums::SDK)
if(BUILD_TESTING)
  add_test(NAME drums_sdk_example COMMAND mr_timestretch_drums_example)
endif()
install(FILES cmake/MRTimestretchDrumsConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MRTimestretchDrums)
